class TaskSchedular{

    PriorityQueue<ScheduledTask> queue = new PriorityQueue<>();
    void schedule(ScheduledTask task){
        queue.offer(task);
        notifyAll();
    }
    void cancelTask(){
        //
    }

    ScheduledTask getNextCandidate(){
        if(queue.isEmpty()){
            return null;
        }else{
            return queue.peek();
        }
    }

    ScheduledTask getTask(){
        if(queue.isEmpty()){
            return null;
        }else{
            return queue.poll();
        }
    }
}

// Dispatcher --> 
class DispatherWorker implements Runnable{
    TaskSchedular taskSchedular;
    public void run(){
        while (true) {

            ScheduledTask scheduledTask;
    
            synchronized (this) {
    
                while (queue.isEmpty()) {
    
                    try {
                        wait();
                    } catch (InterruptedException e) {
                        return;
                    }
                }
    
                scheduledTask = queue.peek();
    
                long currentTime =
                        System.currentTimeMillis();
    
                long waitTime =
                        scheduledTask.executionTime
                        - currentTime;
    
                if (waitTime > 0) {
    
                    try {
                        wait(waitTime); // 
                    } catch (InterruptedException e) {
                        return;
                    }
    
                    continue;
                }
    
                scheduledTask =  queue.poll();
            }
    
            if (scheduledTask.recurring) {
    
                scheduledTask.executionTime =
                        System.currentTimeMillis()
                        + scheduledTask.interval;
    
                taskSchedular.schedule(scheduledTask);
            }
        }
    }
}
