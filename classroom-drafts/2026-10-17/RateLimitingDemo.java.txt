interface RateLimitingStrategy{
    boolean isAllowed(Request request);
}

class TokenBucket{
    int capacity;
    int tokens;
    Long lastRefillTime;
    int refillRate;

    void refill(){
        Long currentTime = Time.currentTime;
        Long diff = lastRefillTime - currentTime;
        tokens = Math.min(capacity, diff * refillRate);
    }

    synchronized boolean isAllowed(){
        refill();

        if(tokens > 0){
            tokens--;
            return true;
        }else{
            return false;
        }
    }
}
