import java.util.List;
import java.util.concurrent.atomic.AtomicInteger;

interface LoadBalancingStrategy{
    Server AssignServer(List<Server> servers, Request request);
}





class RoundRobinStrategy implements  LoadBalancingStrategy{
    AtomicInteger counter = new AtomicInteger(0); //Thread Safe counter 
    @Override
    public Server AssignServer(List<Server> servers, Request request) {
        int index = counter.getAndIncrement() % servers.size(); //
        return servers.get(index); 
    }
}

class LeastConnectionStrategy implements  LoadBalancingStrategy{

    @Override
    public Server AssignServer(List<Server> servers, Request request) {
        int leastConnection = Integer.MAX_VALUE;
        Server ans_server = null;
        for(Server server : servers){
            if(leastConnection > servers.active_connections){
                leastConnection = servers.active_connections;
                ans_server = server;
            }
        }
        return ans_server;
    }
    
}
